CMAKE_MINIMUM_REQUIRED(VERSION 4.0)
PROJECT(FreeImagerip)
INCLUDE(ExternalProject)

IF(NOT CMAKE_BUILD_TYPE)
    SET(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
ENDIF()

EXTERNALPROJECT_ADD(zlib-ng
    URL https://github.com/zlib-ng/zlib-ng/archive/refs/tags/$ENV{ZLIBNG_VER}.tar.gz
    CMAKE_ARGS
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DBUILD_SHARED_LIBS=OFF
        "-DCMAKE_MSVC_RUNTIME_LIBRARY=MultiThreaded$<$<CONFIG:Debug>:Debug>DLL"
        -DZLIB_COMPAT=ON
        -DWITH_GTEST=OFF
        -DZLIBNG_ENABLE_TESTS=OFF
        -DCMAKE_INSTALL_PREFIX=${CMAKE_CURRENT_BINARY_DIR}
)

EXTERNALPROJECT_ADD(libjpeg-turbo
    URL https://github.com/libjpeg-turbo/libjpeg-turbo/archive/refs/tags/$ENV{JPEGTURBO_VER}.tar.gz
    CMAKE_ARGS
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DENABLE_SHARED=OFF
        -DWITH_CRT_DLL=ON
        -DWITH_SIMD=ON
        -DWITH_TESTS=OFF
        -DWITH_TOOLS=OFF
        -DWITH_TURBOJPEG=OFF
        -DCMAKE_INSTALL_PREFIX=${CMAKE_CURRENT_BINARY_DIR}
)

EXTERNALPROJECT_ADD(freeimage
    DEPENDS zlib-ng libjpeg-turbo
    URL https://github.com/danoli3/FreeImage/archive/refs/tags/$ENV{FREEIMAGE_VER}.tar.gz
    CMAKE_ARGS
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DBUILD_SHARED_LIBS=ON
        -DFREEIMAGE_MSVC_RUNTIME=MD
        -DBUILD_LIBHEIF=OFF
        -DBUILD_LIBRAWLITE=OFF
        -DBUILD_LIBJXL=ON
        -DUSE_SYSTEM_ZLIB=ON
        -DUSE_LIBJPEG_TURBO=ON
        -DUSE_SYSTEM_LIBPNG=OFF
        -DUSE_SYSTEM_LIBTIFF=OFF
        -DUSE_SYSTEM_LIBOPENJPEG=OFF
        -DUSE_SYSTEM_WEBP=OFF
        -DUSE_SYSTEM_OPENEXR=OFF
        -DUSE_SYSTEM_JXR=OFF
        -DCMAKE_INSTALL_PREFIX=${CMAKE_CURRENT_BINARY_DIR}
        -DZLIB_INCLUDE_DIR=${CMAKE_CURRENT_BINARY_DIR}/include
        -DZLIB_LIBRARY=${CMAKE_CURRENT_BINARY_DIR}/lib/zlibstatic.lib
        -DJPEG_INCLUDE_DIR=${CMAKE_CURRENT_BINARY_DIR}/include
        -DJPEG_LIBRARY=${CMAKE_CURRENT_BINARY_DIR}/lib/jpeg-static.lib
)

FILE(GLOB FI_EXT_FILES ${CMAKE_CURRENT_SOURCE_DIR}/plugin/FIShellExt/*.cc)
ADD_LIBRARY(FIShellExt SHARED ${FI_EXT_FILES} ${CMAKE_CURRENT_SOURCE_DIR}/plugin/FIShellExt/ShellExt.def)
ADD_DEFINITIONS(-DUNICODE -D_USRDLL)
SET_PROPERTY(TARGET FIShellExt PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
TARGET_INCLUDE_DIRECTORIES(FIShellExt PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/include)
TARGET_LINK_LIBRARIES(FIShellExt ${CMAKE_CURRENT_BINARY_DIR}/lib/FreeImage.lib)
